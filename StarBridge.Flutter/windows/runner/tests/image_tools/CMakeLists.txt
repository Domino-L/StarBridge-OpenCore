cmake_minimum_required(VERSION 3.14)
project(menu_image_tools_tests LANGUAGES CXX)
enable_testing()
get_filename_component(FLUTTER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(EPHEMERAL "${FLUTTER_ROOT}/windows/flutter/ephemeral")
add_executable(menu_image_tools_test ../menu_image_tools_test.cpp ../../menu_local_tools.cpp)
target_include_directories(menu_image_tools_test PRIVATE ../.. "${EPHEMERAL}/cpp_client_wrapper/include")
target_compile_features(menu_image_tools_test PRIVATE cxx_std_17)
target_compile_definitions(menu_image_tools_test PRIVATE NOMINMAX UNICODE _UNICODE STARBRIDGE_MENU_IMAGE_TEST)
target_compile_options(menu_image_tools_test PRIVATE /W4 /WX /wd4100 /utf-8 /EHsc)
target_link_libraries(menu_image_tools_test PRIVATE dwmapi user32 gdi32 ole32 shell32 gdiplus uuid)
add_test(NAME menu_image_tools COMMAND menu_image_tools_test)
set_tests_properties(menu_image_tools PROPERTIES TIMEOUT 15)
