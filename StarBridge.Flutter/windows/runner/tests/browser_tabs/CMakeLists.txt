cmake_minimum_required(VERSION 3.14)
project(menu_browser_tabs_tests LANGUAGES CXX)
enable_testing()
get_filename_component(FLUTTER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(EPHEMERAL "${FLUTTER_ROOT}/windows/flutter/ephemeral")
set(WEBVIEW_SDK "$ENV{USERPROFILE}/.nuget/packages/microsoft.web.webview2/1.0.3179.45")
if(NOT EXISTS "${WEBVIEW_SDK}/build/native/include/WebView2.h")
  message(FATAL_ERROR "Requires the existing local WebView2 SDK; never downloads it")
endif()
add_executable(menu_browser_tabs_test ../menu_browser_tabs_test.cpp ../../menu_local_tools.cpp)
target_include_directories(menu_browser_tabs_test PRIVATE ../.. "${EPHEMERAL}/cpp_client_wrapper/include" "${WEBVIEW_SDK}/build/native/include")
target_compile_features(menu_browser_tabs_test PRIVATE cxx_std_17)
target_compile_definitions(menu_browser_tabs_test PRIVATE NOMINMAX UNICODE _UNICODE STARBRIDGE_HAS_HANGAR_WEBVIEW STARBRIDGE_MENU_BROWSER_TEST)
target_compile_options(menu_browser_tabs_test PRIVATE /W4 /WX /wd4100 /utf-8 /EHsc)
target_link_libraries(menu_browser_tabs_test PRIVATE "${WEBVIEW_SDK}/build/native/x64/WebView2LoaderStatic.lib" dwmapi user32 gdi32 ole32 oleaut32 shell32 gdiplus uuid shlwapi version)
# Also compile the optional-SDK-free path without producing a product binary.
add_library(menu_browser_without_sdk OBJECT ../../menu_local_tools.cpp)
target_include_directories(menu_browser_without_sdk PRIVATE ../.. "${EPHEMERAL}/cpp_client_wrapper/include")
target_compile_features(menu_browser_without_sdk PRIVATE cxx_std_17)
target_compile_definitions(menu_browser_without_sdk PRIVATE NOMINMAX UNICODE _UNICODE)
target_compile_options(menu_browser_without_sdk PRIVATE /W4 /WX /wd4100 /utf-8 /EHsc)
add_test(NAME menu_browser_tabs COMMAND menu_browser_tabs_test)
set_tests_properties(menu_browser_tabs PROPERTIES TIMEOUT 45 WORKING_DIRECTORY "${FLUTTER_ROOT}/..")
